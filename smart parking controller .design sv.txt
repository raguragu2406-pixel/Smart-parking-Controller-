module smart_parking_controller #(
    parameter CAPACITY = 8
)(
    input wire clk,
    input wire reset,
    input wire vehicle_entry,
    input wire vehicle_exit,

    output reg [3:0] occupied_count,
    output wire [3:0] slots_available,
    output wire parking_full,
    output wire entry_allowed,
    output wire exit_allowed
);

always @(posedge clk or posedge reset) begin
    if (reset) begin
        occupied_count <= 4'd0;
    end
    else begin
        if (vehicle_entry && !vehicle_exit &&
            occupied_count < CAPACITY) begin
            occupied_count <= occupied_count + 1'b1;
        end
        else if (vehicle_exit && !vehicle_entry &&
                 occupied_count > 0) begin
            occupied_count <= occupied_count - 1'b1;
        end
    end
end

assign slots_available = CAPACITY - occupied_count;
assign parking_full = (occupied_count == CAPACITY);
assign entry_allowed = (occupied_count < CAPACITY);
assign exit_allowed = (occupied_count > 0);

endmodule
